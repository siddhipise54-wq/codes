def display(board):
    for i in range(3):
        print(board[i * 3], "|", board[i * 3 + 1], "|", board[i * 3 + 2])
        if i < 2:
            print("---------")

def winner(board):
    win = [(0,1,2), (3,4,5), (6,7,8),
           (0,3,6), (1,4,7), (2,5,8),
           (0,4,8), (2,4,6)]

    for a, b, c in win:
        if board[a] == board[b] == board[c] and board[a] != " ":
            return board[a]
    return None

def heuristic(board):
    if winner(board) == "O":
        return 10
    if winner(board) == "X":
        return -10
    return 0

def astar(board):
    best_move = -1
    best_score = -1000

    for i in range(9):
        if board[i] == " ":
            board[i] = "O"
            score = heuristic(board)

            if score == 0:
                for j in range(9):
                    if board[j] == " ":
                        board[j] = "X"
                        if winner(board) == "X":
                            score = -10
                        board[j] = " "

            if score > best_score:
                best_score = score
                best_move = i

            board[i] = " "

    return best_move

def heuristic(state):

def valid(m, c):
    if m < 0 or c < 0 or m > 3 or c > 3:
        return False
    if m > 0 and m < c:
        return False
    if (3 - m) > 0 and (3 - m) < (3 - c):
        return False
    return True


def astar():
    start = (3, 3, 1)
    goal = (0, 0, 0)

    open_list = [(heuristic(start), 0, start, [])]
    visited = []

    moves = [(1, 0), (2, 0), (0, 1),
             (0, 2), (1, 1)]

    while open_list:
        open_list.sort()
        f, g, state, path = open_list.pop(0)

        if state in visited:
            continue

        visited.append(state)
        path = path + [state]

        if state == goal:
            print("Solution Path:")
            for s in path:
                print(s)
            print("\nState Graph:")
            for i in range(len(path) - 1):
                print(path[i], "-->", path[i + 1])
            print("\nTotal steps:", len(path) - 1)
            return

        m, c, boat = state

        for dm, dc in moves:
            if boat == 1:
                new_state = (m - dm, c - dc, 0)
            else:
                new_state = (m + dm, c + dc, 1)

            if valid(new_state[0], new_state[1]):
                if new_state not in visited:
                    new_g = g + 1
                    new_f = new_g + heuristic(new_state)
                    open_list.append(
                        (new_f, new_g, new_state, path)
                    )

    print("No solution found")

def heuristic(state):
    goal = [1, 2, 3, 4, 5, 6, 7, 8, 0]
    count = 0

    for i in range(9):
        if state[i] != 0 and state[i] != goal[i]:
            count += 1

    return count


def astar():
    start = [1, 2, 3, 4, 0, 6, 7, 5, 8]
    goal = [1, 2, 3, 4, 5, 6, 7, 8, 0]

    open_list = [(heuristic(start), 0, start, [])]
    visited = []

    while open_list:
        open_list.sort(key=lambda x: x[0])
        f, g, state, path = open_list.pop(0)

        if state in visited:
            continue

        visited.append(state)
        path = path + [state]

        if state == goal:
            print("Solution Path:")
            for s in path:
                print(s[0:3])
                print(s[3:6])
                print(s[6:9])
                print()
            print("Total steps:", len(path) - 1)
            return

        blank = state.index(0)
        row = blank // 3
        col = blank % 3

        moves = []

        if row > 0:
            moves.append(blank - 3)
        if row < 2:
            moves.append(blank + 3)
        if col > 0:
            moves.append(blank - 1)
        if col < 2:
            moves.append(blank + 1)

        for pos in moves:
            new_state = state.copy()
            new_state[blank] = new_state[pos]
            new_state[pos] = 0

            if new_state not in visited:
                new_g = g + 1
                new_f = new_g + heuristic(new_state)
                open_list.append(
                    (new_f, new_g, new_state, path)
                )

    print("No solution found")

# Alpha-Beta Pruning Algorithm

def alphabeta(tree, level, index, maximizing, alpha, beta):

    # Leaf node
    if level == 2:
        print("Visited leaf:", tree[index])
        return tree[index]

    if maximizing:
        best = -999

        for i in range(2):
            value = alphabeta(tree, level + 1,
                              index * 2 + i, False,
                              alpha, beta)

            best = max(best, value)
            alpha = max(alpha, best)

            if beta <= alpha:
                print("Pruned a branch at MAX node")
                break

        return best

    else:
        best = 999

        for i in range(2):
            value = alphabeta(tree, level + 1,
                              index * 2 + i, True,
                              alpha, beta)

            best = min(best, value)
            beta = min(beta, best)

            if beta <= alpha:
                print("Pruned a branch at MIN node")
                break

        return best


# Leaf values from left to right
tree = [3, 5, 2, 9]

print("Game tree leaf values:", tree)

result = alphabeta(tree, 0, 0, True, -999, 999)

print("Best value:", result)

# Syntax Analysis of Assertive English Statements

text = input("Enter an assertive sentence: ")

# Stage 1: Sentence Segmentation
sentences = text.split(".")
print("\n1. Sentence Segmentation:")
print(sentences)

for sentence in sentences:
    if sentence.strip() == "":
        continue

    # Stage 2: Word Tokenization
    words = sentence.split()
    print("\n2. Word Tokenization:")
    print(words)

    # Stage 3: POS Tagging
    print("\n3. POS Tagging:")
    for word in words:
        w = word.lower().strip(",!?")

        if w in ["the", "a", "an"]:
            print(word, "- Determiner")
        elif w in ["is", "am", "are", "was", "were", "reads", "plays", "eats", "likes"]:
            print(word, "- Verb")
        elif w in ["he", "she", "they", "we", "i"]:
            print(word, "- Pronoun")
        elif w in ["good", "beautiful", "big", "small"]:
            print(word, "- Adjective")
        else:
            print(word, "- Noun")

    # Stage 4: Simple Syntactic Parsing
    print("\n4. Syntactic Parsing:")
    print("Sentence")
    print(" |")
    print(" +-- Subject:", words[0])
    if len(words) > 1:
        print(" +-- Predicate:", " ".join(words[1:]))

print("\nSyntax Analysis Completed")

# Hill Climbing Algorithm

def hill_climbing():
    # Initial state
    current = 0

    # Values of neighboring states
    values = [1, 3, 5, 7, 6, 4, 2]

    print("Hill Climbing Search")

    while True:
        print("Current state:", current,
              "Value:", values[current])

        # Check left neighbor
        left = current - 1
        right = current + 1

        best = current

        if left >= 0 and values[left] > values[best]:
            best = left

        # Check right neighbor
        if right < len(values) and values[right] > values[best]:
            best = right

        # Stop if no better neighbor exists
        if best == current:
            print("Maximum value found:", values[current])
            print("Position:", current)
            break

        current = best


hill_climbing()

# Predicate Logic using PROLOG Concepts

# Facts
parent = [
    ("john", "mary"),
    ("john", "peter"),
    ("mary", "sam"),
    ("peter", "lisa")
]

male = ["john", "peter", "sam"]
female = ["mary", "lisa"]

# Rules
def father(x, y):
    return (x, y) in parent and x in male

def mother(x, y):
    return (x, y) in parent and x in female

def child(x, y):
    return (y, x) in parent

def sibling(x, y):
    for p1 in parent:
        for p2 in parent:
            if p1[0] == p2[0] and p1[1] == x and p2[1] == y and x != y:
                return True
    return False

# Queries
print("Facts:")
print("Parent:", parent)
print("Male:", male)
print("Female:", female)

print("\nQueries:")
print("Is John the father of Mary?", father("john", "mary"))
print("Is Mary the mother of Sam?", mother("mary", "sam"))
print("Is Sam the child of Mary?", child("sam", "mary"))
print("Are Mary and Peter siblings?", sibling("mary", "peter"))
print("Are Mary and Peter siblings?", sibling("mary", "peter"))
